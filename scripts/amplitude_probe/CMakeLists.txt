cmake_minimum_required(VERSION 3.30)
cmake_policy(SET CMP0091 NEW)
project(AmplitudeImportProbe LANGUAGES CXX)

  message(FATAL_ERROR "Provide DAE_ROOT and the installed AM_SDK_PATH")

include("${DAE_ROOT}/cmake/Amplitude.cmake")
darkangel_import_amplitude("${AM_SDK_PATH}")

target_link_libraries(AmplitudeImportProbe PRIVATE DarkAngel::AmplitudeSDK)
