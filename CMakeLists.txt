cmake_minimum_required(VERSION 3.30)
cmake_policy(SET CMP0091 NEW)
project(DarkAngel VERSION 0.0.1 LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "M0 requires Windows x64/MSVC. Other hosts do not validate this baseline.")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set(CMAKE_CXX_EXTENSIONS OFF)
option(DAE_SHIPPING "Exclude authoring and agent capabilities" OFF)
option(DAE_AUTHORING "Enable native authoring capabilities" OFF)
option(DAE_AGENT_ENDPOINTS "Enable scoped editor agent endpoints" OFF)
option(DAE_ASSETS "Build the native M2 asset tools" OFF)
option(DAE_RENDERER_SPIKE "Build Diligent device/editor acceptance targets" OFF)
option(DAE_ANIMATION "Build frozen Ozz runtime and owned animation cooks" OFF)
option(DAE_PHYSICS "Build pinned Jolt motor adapter" OFF)
option(DAE_NETWORK_GNS "Build the offline GNS adapter" OFF)
option(DAE_NETWORK_EOS "Build the explicit online EOS adapter" OFF)
set(DAE_EOS_SDK_ROOT "" CACHE PATH "Authorized pinned EOS Windows C SDK root")
if(DAE_SHIPPING AND (DAE_AUTHORING OR DAE_AGENT_ENDPOINTS OR DAE_ASSETS OR DAE_RENDERER_SPIKE))
  message(FATAL_ERROR "Shipping excludes authoring and agent endpoints")
endif()
if(DAE_AGENT_ENDPOINTS AND NOT DAE_AUTHORING)
  message(FATAL_ERROR "Agent endpoints require an explicit authoring build")
endif()
include(cmake/test_helpers.cmake)
include(cmake/engine.cmake)
include(CTest)
include(cmake/authoring.cmake)
include(cmake/assets.cmake)
include(cmake/editor.cmake)
include(cmake/tests.cmake)
include(cmake/install.cmake)
