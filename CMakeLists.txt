cmake_minimum_required(VERSION 3.30)
cmake_policy(SET CMP0091 NEW)
project(DarkAngel VERSION 0.0.1 LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "M0 requires Windows x64/MSVC. Other hosts do not validate this baseline.")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set(CMAKE_CXX_EXTENSIONS OFF)
option(DAE_SHIPPING "Exclude authoring and agent capabilities" OFF)
option(DAE_AUTHORING "Enable authoring capabilities in later targets" OFF)
option(DAE_AGENT_ENDPOINTS "Enable agent endpoints in later targets" OFF)
if(DAE_SHIPPING AND (DAE_AUTHORING OR DAE_AGENT_ENDPOINTS))
  message(FATAL_ERROR "Shipping excludes authoring and agent endpoints")
endif()
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" rev-parse --short=12 HEAD WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}" OUTPUT_VARIABLE DAE_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(NOT DAE_REVISION)
  set(DAE_REVISION "uncommitted-m0")
endif()
execute_process(COMMAND "${GIT_EXECUTABLE}" status --porcelain WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}" OUTPUT_VARIABLE DAE_DIRTY ERROR_QUIET)
if(DAE_DIRTY)
  string(APPEND DAE_REVISION "-dirty")
endif()
configure_file(cmake/build_info.hpp.in generated/darkangel/build_info.hpp @ONLY)
add_library(DarkAngelFoundation STATIC engine/foundation/build_info.cpp)
target_compile_features(DarkAngelFoundation PUBLIC cxx_std_20)
target_include_directories(DarkAngelFoundation PUBLIC engine/foundation/include PRIVATE "${PROJECT_BINARY_DIR}/generated")
add_library(DarkAngelRuntime STATIC engine/runtime/bootstrap.cpp)
target_compile_features(DarkAngelRuntime PUBLIC cxx_std_20)
target_include_directories(DarkAngelRuntime PUBLIC engine/runtime/include)
target_link_libraries(DarkAngelRuntime PRIVATE DarkAngelFoundation)
add_executable(DarkAngelHeadless apps/headless/main.cpp)
target_link_libraries(DarkAngelHeadless PRIVATE DarkAngelRuntime)
foreach(target DarkAngelFoundation DarkAngelRuntime DarkAngelHeadless)
  target_compile_options(${target} PRIVATE /W4 /utf-8)
endforeach()
include(CTest)
if(BUILD_TESTING)
  find_package(flecs CONFIG REQUIRED)
  find_package(unofficial-luau CONFIG REQUIRED)
  find_package(unofficial-sqlite3 CONFIG REQUIRED)
  add_executable(FlecsProbe tests/flecs_probe.cpp)
  target_link_libraries(FlecsProbe PRIVATE flecs::flecs_static)
  add_executable(LuauProbe tests/luau_probe.cpp)
  target_link_libraries(LuauProbe PRIVATE unofficial::luau::Luau.Compiler unofficial::luau::Luau.VM)
  add_executable(SQLiteProbe tests/sqlite_probe.cpp)
  target_link_libraries(SQLiteProbe PRIVATE unofficial::sqlite3::sqlite3)
  foreach(target FlecsProbe LuauProbe SQLiteProbe)
    target_compile_features(${target} PRIVATE cxx_std_20)
    target_compile_options(${target} PRIVATE /W4 /utf-8)
    add_test(NAME ${target} COMMAND ${target})
    set_tests_properties(${target} PROPERTIES TIMEOUT 10 LABELS "M0;dependency")
  endforeach()
  add_test(NAME HeadlessSmoke COMMAND DarkAngelHeadless)
  set_tests_properties(HeadlessSmoke PROPERTIES TIMEOUT 5 PASS_REGULAR_EXPRESSION "DarkAngel 0.0.1")
endif()
include(cmake/install.cmake)
